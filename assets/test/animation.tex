% Test presentation for TeXora's animation support: two animations made with the animate package.
% Rebuild with: pdflatex animation.tex (twice).
\documentclass{beamer}
\usepackage{animate}
\usepackage{tikz}
\begin{document}
\begin{frame}{Autoplay and loop}
  \begin{animateinline}[autoplay,loop,controls]{4}
    \multiframe{6}{rX=0+1}{%
      \begin{tikzpicture}
        \draw (0,-0.5) rectangle (5.5,0.5);
        \fill[green!50!black] (\rX,0) circle (0.4);
      \end{tikzpicture}}
  \end{animateinline}
\end{frame}
\begin{frame}{Back and forth, started by hand}
  \begin{animateinline}[palindrome,controls]{2}
    \multiframe{4}{rY=0+1}{%
      \begin{tikzpicture}
        \draw (-0.5,-0.5) rectangle (0.5,3.5);
        \fill[red!70!black] (0,\rY) circle (0.4);
      \end{tikzpicture}}
  \end{animateinline}
\end{frame}
\end{document}
